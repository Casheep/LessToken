cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" LESSTOKEN_BUILD_VERSION LIMIT_COUNT 1)
project(LessToken VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(lesstoken src/main.cpp src/core.cpp src/storage.cpp src/process.cpp src/mcp.cpp src/hook.cpp src/prompt.cpp src/query.cpp src/records.cpp src/batch.cpp)
target_include_directories(lesstoken PRIVATE include third_party)
target_compile_definitions(lesstoken PRIVATE LESSTOKEN_VERSION="${LESSTOKEN_BUILD_VERSION}")
if(WIN32)
  target_compile_definitions(lesstoken PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(lesstoken PRIVATE bcrypt shell32)
endif()
if(MSVC)
  set_property(TARGET lesstoken PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(lesstoken PRIVATE /W4 /utf-8)
else()
  target_compile_options(lesstoken PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(lesstoken PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
endif()
if(BUILD_TESTING AND Python3_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/tests/optimization.py")
  add_test(NAME optimization COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/optimization.py $<TARGET_FILE:lesstoken>)
endif()
if(BUILD_TESTING AND Python3_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/tests/session_usage.py")
  add_test(NAME session_usage COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/session_usage.py)
endif()
if(BUILD_TESTING AND Python3_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/tests/evaluation.py")
  add_test(NAME evaluation COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/evaluation.py)
endif()
if(BUILD_TESTING AND Python3_FOUND AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND EXISTS "${CMAKE_SOURCE_DIR}/tests/coding_evaluation.py")
  add_test(NAME coding_evaluation COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/coding_evaluation.py ${CMAKE_CXX_COMPILER})
endif()
if(BUILD_TESTING AND Python3_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/tests/prompt_prefetch.py")
  add_test(NAME prompt_prefetch COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/prompt_prefetch.py $<TARGET_FILE:lesstoken>)
endif()
if(BUILD_TESTING AND Python3_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/tests/hooks.py")
  add_test(NAME hooks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/hooks.py $<TARGET_FILE:lesstoken>)
endif()
if(BUILD_TESTING AND Python3_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/tests/publication.py")
  add_test(NAME publication COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/publication.py)
endif()
if(BUILD_TESTING AND Python3_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/tests/integration.py")
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:lesstoken>)
  add_test(NAME cpp_port COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/port.py $<TARGET_FILE:lesstoken>)
endif()
